   191→   191→  const normalizedSites = selectedSites.map(normalizeExamSiteValue).filter(Boolean);

   194→   194→    return selectedRows.reduce((sum, row) => sum + Number(row.activeStudents || 0), 0);
   195→   195→  }

   197→   197→    return sum + normalizedSites.reduce((siteSum, site) => siteSum + Number(row.siteCounts?.[site] || 0), 0);
   198→   198→  }, 0);
