    75→function buildJudgmentPreview(state: ExamFormState): JudgmentPreviewItem[] {
    76→  const fullMark = Number(state.fullMark || 0);
    77→  const passMark = Number(state.passMark || 0);
    78→  const discountMark = Number(state.discountMark || 0);
    79→  const penalty = Number(state.opportunitiesPenaltyNum || 0);
    80→  const isFinalExam = state.type === "فاينل";
    81→  const noDiscount = Boolean(state.noDiscount);
    82→  const dismissalGradeRaw = state.dismissalGrade === "" ? null : Number(state.dismissalGrade);
    83→  const dismissalGrade = Number.isFinite(Number(dismissalGradeRaw)) ? Number(dismissalGradeRaw) : null;
    84→  const items: JudgmentPreviewItem[] = [];
    85→
    86→  if (noDiscount) {
    87→    items.push({
    88→      title: `الدرجات من 0 إلى ${formatRangeNumber(fullMark)}`,
    89→      description: `لا تخصم فرص. من ${formatRangeNumber(passMark)} فما فوق تظهر ناجح، وأقل من ذلك تظهر بدون خصم.`,
    90→      tone: "info",
    91→    });
    92→    items.push({ title: "الغياب", description: "لا يخصم فرص لأن الامتحان بدون خصم.", tone: "info" });
    93→    items.push({ title: "الغش", description: "يبقى إجراءً خطيراً: أول غش يفصل مؤقتاً ويصفر الفرص، والغش المتكرر يفصل نهائياً.", tone: "danger" });
    94→    return items;
    95→  }
    96→
    97→  if (isFinalExam) {
    98→    if (dismissalGrade !== null) {
    99→      items.push({ title: `من 0 إلى ${formatRangeNumber(dismissalGrade)}`, description: "فصل مؤقت حسب درجة الفصل في الفاينل.", tone: "danger" });
   100→      if (dismissalGrade >= passMark) {
   101→        items.push({ title: "تنبيه تداخل", description: "درجة الفصل تساوي أو تتجاوز درجة النجاح، وهذا يجعل حكم الفصل يتداخل مع النجاح. راجع القيم قبل الحفظ.", tone: "danger" });
   102→      }
   103→      items.push({ title: `أكبر من ${formatRangeNumber(dismissalGrade)} وأقل من ${formatRangeNumber(passMark)}`, description: "راسب في الفاينل بدون خصم فرص مباشر.", tone: "warn" });
   104→    } else {
   105→      items.push({ title: "درجة 0", description: "فصل مؤقت في الفاينل.", tone: "danger" });
   106→      items.push({ title: `أكبر من 0 وأقل من ${formatRangeNumber(passMark)}`, description: "راسب.", tone: "warn" });
   107→    }
   108→    items.push({ title: `من ${formatRangeNumber(passMark)} فما فوق`, description: "ناجح.", tone: "ok" });
   109→    items.push({ title: "الغياب", description: "فصل مؤقت لأنه غياب ضمن فاينل.", tone: "danger" });
   110→    items.push({ title: "الغش", description: "أول غش يفصل مؤقتاً ويصفر الفرص، والغش المتكرر يفصل نهائياً.", tone: "danger" });
   111→    return items;
   112→  }
   113→
   114→  items.push({ title: `من 0 إلى ${formatRangeNumber(discountMark)}`, description: `مخصوم: يخصم ${formatRangeNumber(penalty)} فرصة من الطالب.`, tone: "danger" });
   115→  items.push({ title: `أكبر من ${formatRangeNumber(discountMark)} وأقل من ${formatRangeNumber(passMark)}`, description: "راسب/محاسبة رسوب بدون خصم فرص مباشر.", tone: "warn" });
   116→  items.push({ title: `من ${formatRangeNumber(passMark)} فما فوق`, description: "ناجح.", tone: "ok" });
   117→  items.push({ title: "الغياب", description: `مخصوم: يخصم ${formatRangeNumber(penalty)} فرصة.`, tone: "danger" });
   118→  items.push({ title: "الغش", description: "أول غش يفصل مؤقتاً ويصفر الفرص، والغش المتكرر يفصل نهائياً.", tone: "danger" });
   119→  return items;
   120→}

   153→    return { active: true, scheduledActivateAt: "", scheduledDeactivateAt: "" };
   154→  }

   161→function buildExamPayload(form: ExamFormState): Omit<Exam, "id"> {
   162→  const isFinalExam = form.type === "فاينل";
   163→  const noDiscount = Boolean(form.noDiscount);
   164→  return {
   165→    name: form.name.trim(),
   166→    type: form.type,
   167→    courseIds: form.courseIds,
   168→    mainSite: form.mainSites.join(","),
   169→    date: form.date,
   170→    fullMark: form.fullMark,
   171→    passMark: form.passMark,
   172→    discountMark: isFinalExam || noDiscount ? 0 : form.discountMark,
   173→    opportunitiesPenalty: noDiscount ? 0 : (isFinalExam ? "فصل مؤقت" : form.opportunitiesPenaltyNum),
   174→    dismissalGrade: !noDiscount && isFinalExam && form.dismissalGrade ? Number(form.dismissalGrade) : null,
   175→    noDiscount,
   176→    ...applyStatus(form),
   177→  };
   178→}

   180→function selectedCourseBlockers(rows: ExamCreateContextRow[], selectedIds: string[]): string[] {
   181→  return selectedIds.flatMap((id) => {
   182→    const row = rows.find((item) => item.id === id);
   183→    if (!row) return [`الدورة ${id} غير موجودة في سياق بيانات النظام`];
   184→    return row.canSelectForExam ? [] : row.blockers.map((blocker) => `${String(row.course?.name || row.id)}: ${blocker}`);
   185→  });
   186→}

   188→function selectedSiteActiveStudentCount(rows: ExamCreateContextRow[], selectedCourseIds: string[], selectedSites: string[]): number | null {
   189→  if (selectedCourseIds.length === 0 || selectedSites.length === 0) return null;
   190→  const selectedRows = rows.filter((row) => selectedCourseIds.includes(row.id));
   191→  const normalizedSites = selectedSites.map(normalizeExamSiteValue).filter(Boolean);
   192→  const allSelected = EXAM_MAIN_SITE_OPTIONS.every((site) => normalizedSites.includes(normalizeExamSiteValue(site)));
   193→  if (allSelected) {
   194→    return selectedRows.reduce((sum, row) => sum + Number(row.activeStudents || 0), 0);
   195→  }
   196→  return selectedRows.reduce((sum, row) => {
   197→    return sum + normalizedSites.reduce((siteSum, site) => siteSum + Number(row.siteCounts?.[site] || 0), 0);
   198→  }, 0);
   199→}

   255→    const noDiscount = Boolean(state.noDiscount);
   256→    const discountMark = isFinalExam || noDiscount
   257→      ? 0
   258→      : Number(state.discountMark);
   259→    const opportunitiesPenalty = Number(state.opportunitiesPenaltyNum);
   260→    const dismissalGrade = state.dismissalGrade === "" ? null : Number(state.dismissalGrade);
   261→
   262→    if (contextLoading) return "انتظر تحميل سياق إضافة الامتحان من بيانات النظام";
   263→    if (contextError) return contextError;
   264→    if (!state.name.trim()) return "يرجى إدخال اسم الامتحان";
   265→    if (state.courseIds.length === 0) return "يرجى اختيار دورة واحدة على الأقل";
